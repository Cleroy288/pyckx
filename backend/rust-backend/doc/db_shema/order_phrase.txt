-- Order Phrase Game Tables
-- Run this migration in Supabase SQL Editor

-- ============================================================================
-- Order Phrase Sets table
-- ============================================================================
CREATE TABLE IF NOT EXISTS intello_order_phrase_sets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    description TEXT NOT NULL,
    level TEXT NOT NULL CHECK (level IN ('easy', 'medium', 'hard')),
    language TEXT NOT NULL DEFAULT 'en',
    subjects TEXT[] NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================================
-- Order Phrase Questions table (individual phrase puzzles)
-- ============================================================================
CREATE TABLE IF NOT EXISTS intello_order_phrase_questions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    set_id UUID NOT NULL REFERENCES intello_order_phrase_sets(id) ON DELETE CASCADE,
    original_phrase TEXT NOT NULL,
    hint TEXT NOT NULL DEFAULT '',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================================
-- Order Phrase Words table (words with their correct positions)
-- ============================================================================
CREATE TABLE IF NOT EXISTS intello_order_phrase_words (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    question_id UUID NOT NULL REFERENCES intello_order_phrase_questions(id) ON DELETE CASCADE,
    word TEXT NOT NULL,
    position INTEGER NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================================
-- Indexes for performance
-- ============================================================================
CREATE INDEX IF NOT EXISTS idx_order_phrase_sets_user_id ON intello_order_phrase_sets(user_id);
CREATE INDEX IF NOT EXISTS idx_order_phrase_questions_set_id ON intello_order_phrase_questions(set_id);
CREATE INDEX IF NOT EXISTS idx_order_phrase_words_question_id ON intello_order_phrase_words(question_id);

-- ============================================================================
-- Row Level Security Policies
-- ============================================================================
ALTER TABLE intello_order_phrase_sets ENABLE ROW LEVEL SECURITY;
ALTER TABLE intello_order_phrase_questions ENABLE ROW LEVEL SECURITY;
ALTER TABLE intello_order_phrase_words ENABLE ROW LEVEL SECURITY;

-- Users can manage their own sets
CREATE POLICY "Users can manage own order phrase sets" ON intello_order_phrase_sets
    FOR ALL USING (auth.uid() = user_id);

-- Users can manage questions in their own sets
CREATE POLICY "Users can manage own order phrase questions" ON intello_order_phrase_questions
    FOR ALL USING (
        EXISTS (SELECT 1 FROM intello_order_phrase_sets WHERE id = set_id AND user_id = auth.uid())
    );

-- Users can manage words in their own questions
CREATE POLICY "Users can manage own order phrase words" ON intello_order_phrase_words
    FOR ALL USING (
        EXISTS (
            SELECT 1 FROM intello_order_phrase_questions q
            JOIN intello_order_phrase_sets s ON q.set_id = s.id
            WHERE q.id = question_id AND s.user_id = auth.uid()
        )
    );
