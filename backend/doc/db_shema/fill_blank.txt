-- Fill in the Blank Game Tables
-- Run this migration in Supabase SQL Editor

-- ============================================================================
-- Fill Blank Sets table
-- ============================================================================
CREATE TABLE IF NOT EXISTS intello_fill_blank_sets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    description TEXT NOT NULL,
    level TEXT NOT NULL CHECK (level IN ('easy', 'medium', 'hard')),
    language TEXT NOT NULL DEFAULT 'en',
    subjects TEXT[] NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================================
-- Fill Blank Questions table
-- ============================================================================
CREATE TABLE IF NOT EXISTS intello_fill_blank_questions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    set_id UUID NOT NULL REFERENCES intello_fill_blank_sets(id) ON DELETE CASCADE,
    phrase TEXT NOT NULL,
    explanation TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================================
-- Fill Blank Options table (answer choices)
-- ============================================================================
CREATE TABLE IF NOT EXISTS intello_fill_blank_options (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    question_id UUID NOT NULL REFERENCES intello_fill_blank_questions(id) ON DELETE CASCADE,
    text TEXT NOT NULL,
    is_correct BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================================
-- Indexes for performance
-- ============================================================================
CREATE INDEX IF NOT EXISTS idx_fill_blank_sets_user_id ON intello_fill_blank_sets(user_id);
CREATE INDEX IF NOT EXISTS idx_fill_blank_questions_set_id ON intello_fill_blank_questions(set_id);
CREATE INDEX IF NOT EXISTS idx_fill_blank_options_question_id ON intello_fill_blank_options(question_id);

-- ============================================================================
-- Row Level Security Policies
-- ============================================================================
ALTER TABLE intello_fill_blank_sets ENABLE ROW LEVEL SECURITY;
ALTER TABLE intello_fill_blank_questions ENABLE ROW LEVEL SECURITY;
ALTER TABLE intello_fill_blank_options ENABLE ROW LEVEL SECURITY;

-- Users can manage their own sets
CREATE POLICY "Users can manage own fill blank sets" ON intello_fill_blank_sets
    FOR ALL USING (auth.uid() = user_id);

-- Users can manage questions in their own sets
CREATE POLICY "Users can manage own fill blank questions" ON intello_fill_blank_questions
    FOR ALL USING (
        EXISTS (SELECT 1 FROM intello_fill_blank_sets WHERE id = set_id AND user_id = auth.uid())
    );

-- Users can manage options in their own questions
CREATE POLICY "Users can manage own fill blank options" ON intello_fill_blank_options
    FOR ALL USING (
        EXISTS (
            SELECT 1 FROM intello_fill_blank_questions q
            JOIN intello_fill_blank_sets s ON q.set_id = s.id
            WHERE q.id = question_id AND s.user_id = auth.uid()
        )
    );
