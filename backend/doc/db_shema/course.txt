-- == COURSE ==

CREATE TABLE intello_courses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    description TEXT NOT NULL DEFAULT '',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- == USER RESOURCES (Single Source of Truth) ==

CREATE TABLE intello_user_resources (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
    filename TEXT NOT NULL,
    content TEXT NOT NULL,
    token_count INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(user_id, filename)
);

-- == COURSE-RESOURCE JUNCTION (Many-to-Many) ==

CREATE TABLE intello_course_resource_links (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL REFERENCES intello_courses(id) ON DELETE CASCADE,
    resource_id UUID NOT NULL REFERENCES intello_user_resources(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, resource_id)
);

-- Indices for performance
CREATE INDEX idx_intello_user_resources_user_id ON intello_user_resources(user_id);
CREATE INDEX idx_intello_course_resource_links_course_id ON intello_course_resource_links(course_id);

-- RLS Policies
ALTER TABLE intello_user_resources ENABLE ROW LEVEL SECURITY;
CREATE POLICY intello_user_resources_owner ON intello_user_resources FOR ALL 
USING (auth.uid() = user_id);

ALTER TABLE intello_course_resource_links ENABLE ROW LEVEL SECURITY;
CREATE POLICY intello_course_resource_links_owner ON intello_course_resource_links FOR ALL 
USING (course_id IN (SELECT id FROM intello_courses WHERE user_id = auth.uid()));

-- == STUDY SESSION ==

CREATE TABLE intello_study_sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL REFERENCES intello_courses(id) ON DELETE CASCADE,
    topic VARCHAR(255) NOT NULL,
    instructions TEXT NOT NULL DEFAULT '',
    keywords TEXT[] DEFAULT '{}',
    language VARCHAR(5) DEFAULT 'en',
    status VARCHAR(20) NOT NULL DEFAULT 'in_progress',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    completed_at TIMESTAMPTZ
);

CREATE TABLE intello_session_content (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    content_type VARCHAR(50) NOT NULL,
    title VARCHAR(255),
    content TEXT NOT NULL,
    mermaid_code TEXT,
    order_index INT DEFAULT 0,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE intello_session_resources (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    resource_id UUID NOT NULL REFERENCES intello_user_resources(id) ON DELETE CASCADE,
    UNIQUE(session_id, resource_id)
);

-- == SESSION GAMES ==

CREATE TABLE intello_session_games (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    game_type VARCHAR(50) NOT NULL,
    game_set_id UUID NOT NULL,
    errors_count INTEGER NOT NULL DEFAULT 0,
    total_questions INTEGER NOT NULL DEFAULT 0,
    completed BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- == Q&A ==

CREATE TABLE intello_session_questions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    section_context TEXT NOT NULL,
    question TEXT NOT NULL,
    answer TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- == NOTES ==

CREATE TABLE intello_session_notes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    game_type VARCHAR(50),
    game_set_id UUID,
    note TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- == CONTENT MODIFICATIONS ==

CREATE TABLE intello_content_modifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    game_type VARCHAR(50) NOT NULL,
    game_set_id UUID NOT NULL,
    modification_type VARCHAR(50) NOT NULL,
    request TEXT,
    result TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- == SYNTHESIS ==

CREATE TABLE intello_session_synthesis (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL REFERENCES intello_study_sessions(id) ON DELETE CASCADE,
    version INTEGER NOT NULL DEFAULT 1,
    content TEXT NOT NULL,
    errors_summary JSONB NOT NULL DEFAULT '{}',
    notes_addressed INTEGER NOT NULL DEFAULT 0,
    combined_from UUID[] DEFAULT '{}',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE intello_synthesis_modifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    synthesis_id UUID NOT NULL REFERENCES intello_session_synthesis(id) ON DELETE CASCADE,
    modification_type VARCHAR(50) NOT NULL,
    request TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
