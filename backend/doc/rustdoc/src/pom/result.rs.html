1use std::fmt::{self, Display};
2use std::error;
3
4/// Parser error.
5#[derive(Debug, PartialEq)]
6pub enum Error {
7	Incomplete,
8	Mismatch { message: String, position: usize },
9	Conversion { message: String, position: usize },
10	Expect { message: String, position: usize, inner: Box<Error> },
11	Custom { message: String, position: usize, inner: Option<Box<Error>> },
12}
13
14impl error::Error for Error {
15	fn description(&self) -> &'static str {
16		"Parse error"
17	}
18}
19
20impl Display for Error {
21	fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
22		match self {
23			&Error::Incomplete =>
24				write!(f, "Incomplete"),
25			&Error::Mismatch { ref message, ref position } =>
26				write!(f, "Mismatch at {}: {}", position, message),
27			&Error::Conversion { ref message, ref position } =>
28				write!(f, "Conversion failed at {}: {}", position, message),
29			&Error::Expect { ref message, ref position, ref inner } =>
30				write!(f, "{} at {}: {}", message, position, inner),
31			&Error::Custom { ref message, ref position, inner: Some(ref inner) } =>
32				write!(f, "{} at {}, (inner: {})", message, position, inner),
33			&Error::Custom { ref message, ref position, inner: None } =>
34				write!(f, "{} at {}", message, position),
35		}
36	}
37}
38
39/// Parser result, `Result<O>` ia alias of `Result<O, pom::Error>`.
40pub type Result<O> = ::std::result::Result<O, Error>;
